# Also usable standalone for helper tests.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.12)
  project(AMuleMediaProbe LANGUAGES CXX)
  option(BUILD_TESTING "Build tests" ON)
  enable_testing()
endif()
# The helper does not link wxWidgets, so it must request large-file stdio itself.
# Apply this to both executables, including standalone helper builds.
if(NOT WIN32)
  add_compile_definitions(_FILE_OFFSET_BITS=64)
endif()
find_library(MEDIAINFO_LIBRARY NAMES mediainfo libmediainfo.so.0)
if(NOT MEDIAINFO_LIBRARY)
  message(FATAL_ERROR "MediaInfoLib required: install libmediainfo-dev or configure -DENABLE_MEDIA_PROBE=OFF")
endif()
find_library(MEDIA_ZEN_LIBRARY NAMES zen libzen.so.0)
if(NOT MEDIA_ZEN_LIBRARY)
  message(FATAL_ERROR "ZenLib required for MediaInfoLib: install libzen-dev or configure -DENABLE_MEDIA_PROBE=OFF")
endif()
add_executable(amule-media-probe main.cpp)
target_compile_features(amule-media-probe PRIVATE cxx_std_17)
target_link_libraries(amule-media-probe PRIVATE ${MEDIAINFO_LIBRARY} ${MEDIA_ZEN_LIBRARY})
if(MEDIA_PROBE_STATIC)
  find_package(ZLIB QUIET)
  if(NOT ZLIB_FOUND)
    message(FATAL_ERROR "zlib required for static MediaInfoLib: install zlib1g-dev or configure -DENABLE_MEDIA_PROBE=OFF")
  endif()
  find_package(Threads QUIET)
  if(NOT Threads_FOUND)
    message(FATAL_ERROR "Thread support required for static ZenLib: install platform thread libraries or configure -DENABLE_MEDIA_PROBE=OFF")
  endif()
  target_link_libraries(amule-media-probe PRIVATE ZLIB::ZLIB Threads::Threads ${CMAKE_DL_LIBS})
endif()
# Same directory as the core, including relocated bundles and portable installs.
set_target_properties(amule-media-probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/src")
install(TARGETS amule-media-probe RUNTIME DESTINATION bin)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME MediaProbeBackends COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/test_backends.py $<TARGET_FILE:amule-media-probe>)
  if(BUILD_TESTING)
    add_executable(media-probe-seek-fixture main.cpp test_seek_stub.cpp)
    target_compile_features(media-probe-seek-fixture PRIVATE cxx_std_17)
    add_test(NAME MediaProbeSeekLimit COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/test_seek_limit.py $<TARGET_FILE:media-probe-seek-fixture>)
  endif()
endif()
install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../packaging/media/licenses/"
  DESTINATION share/licenses/amule-media-probe)
